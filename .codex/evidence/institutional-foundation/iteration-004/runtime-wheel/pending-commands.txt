import hashlib
import json
import os
from email.parser import BytesParser
from pathlib import Path
from zipfile import ZipFile

root = Path(os.environ["RD4_EVIDENCE"])
work = Path(os.environ["RD4_WORK"])
source = json.loads((root / "source-manifest.json").read_text())
assert source["source_commit"] == os.environ["RD4_SOURCE"]
archives = sorted((root / "artifacts").glob("*.whl"))
main = [path for path in archives if path.name.startswith("codex_ml-")]
assert len(main) == 1
assert len([path for path in archives if path.name.startswith("codex_contracts-")]) == 1
inventory = []
for path in archives:
    with ZipFile(path) as archive:
        records = []
        for info in archive.infolist():
            data = archive.read(info)
            records.append({"path": info.filename, "bytes": len(data),
                            "sha256": hashlib.sha256(data).hexdigest()})
        metadata_name = next(name for name in archive.namelist() if name.endswith(".dist-info/METADATA"))
        distribution = BytesParser().parsebytes(archive.read(metadata_name))
        inventory.append({"wheel": path.name, "sha256": hashlib.sha256(path.read_bytes()).hexdigest(),
                          "name": distribution["Name"], "version": distribution["Version"],
                          "requires_dist": distribution.get_all("Requires-Dist", []), "files": records})
(root / "wheel-manifest.json").write_text(json.dumps(inventory, sort_keys=True, indent=2) + "\n")
formulas = [Path(name).name for name in source["focused_source_files"]
            if name.startswith("src/codex/institutional/") and "formula" in name and name.endswith(".json")]
release = {"source_commit": source["source_commit"], "repository_root": os.environ["RD4_REPO"],
           "export_root": str(work / "source-export"), "runtime_cwd": str(work / "external-runtime"),
           "wheel": str(main[0]), "wheel_sha256": hashlib.sha256(main[0].read_bytes()).hexdigest(),
           "formula_resources": formulas,
           "source_files": {name: data["sha256"] for name, data in source["focused_source_files"].items()}}
(root / "release.json").write_text(json.dumps(release, sort_keys=True, indent=2) + "\n")
PY
